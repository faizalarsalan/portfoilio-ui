    bin_dir = str(PROJECT_ROOT / "bin")
    # Ensure FFmpeg and project bin are in PATH
    path_parts = [bin_dir, DEFAULT_FFMPEG_DIR, env.get("PATH", "")]
    env["PATH"] = ";".join(path_parts)
    env["HYPERFRAMES_BROWSER_PATH"] = DEFAULT_BROWSER_PATH
    env["HYPERFRAMES_WHISPER_PATH"] = DEFAULT_WHISPER_PATH
    env["HYPERFRAMES_SKIP_SKILLS"] = "1"

  const [r, g, b] = hex(value).map((v) => {
    const c = v / 255;
    return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;

};

  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return (hi + 0.05) / (lo + 0.05);
